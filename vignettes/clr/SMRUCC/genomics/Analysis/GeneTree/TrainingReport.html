TrainingReport {SMRUCC.genomics.Analysis.GeneTree} .NET clr documentation

TrainingReport

Description

逐基因拟合的训练报告：既可以在控制台打印摘要，也可以写成两张可读的表格文件 （report.tsv 与 importance.tsv），用于人工核查调控网络是否合理。

Declare

            
# namespace SMRUCC.genomics.Analysis.GeneTree
export class TrainingReport {
   # 成功训练出梯度提升树的基因数量
   BoostedModels: integer;
   # 参与训练的细胞数量
   Cells: integer;
   # 本次训练使用的配置
   Config: GeneTreeConfig;
   # 退化为常数模型的基因数量
   ConstantModels: integer;
   # R² 大于 0.05（说明该靶基因确实受到其调控者的非线性影响）的模型数量
   InformativeModels: integer;
   # 所有梯度提升树模型在验证集上的 R² 均值
   MeanR2: double;
   # 所有梯度提升树模型在验证集上的 R² 中位数
   MedianR2: double;
   # 每个基因的模型（下标与表达矩阵的行一致，可能包含 Nothing）
   Models: GeneModel[];
   # 没有任何调控者的源节点数量
   SourceNodes: integer;
}
