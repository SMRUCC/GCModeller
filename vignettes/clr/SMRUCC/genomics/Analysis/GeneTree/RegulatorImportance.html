RegulatorImportance

Description

单个调控因子在一个基因模型之中的重要性统计

Declare

            
# namespace SMRUCC.genomics.Analysis.GeneTree
export class RegulatorImportance {
   # 与靶基因之间的 pearson 相关系数
   correlation: double;
   # 与靶基因之间的互信息
   mi: double;
   # 在特征向量之中的位置（0 起）
   rank: integer;
   # 调控因子编号
   regulator: string;
   # 灵敏度：其余特征取训练集的中位数时，把该特征从最小值扫描到最大值所引起的预测变化幅度。
   # 
   # 这是一个廉价的“偏依赖”近似，用于大致判断该调控边对靶基因的贡献大小与强度。
   sensitivity: double;
   # 所有树之中，以该特征作为分裂特征的节点数量
   splits: integer;
}
