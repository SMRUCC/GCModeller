# namespace SMRUCC.genomics.Analysis.GeneTree
export class ModelPerformance {
   # 用“标签均值”作为预测时得到的均方根误差（也就是标签的标准差）， 用于衡量模型相对常数的改进幅度。
   baselineRmse: double;
   # 每个被选中调控因子的重要性
   importance: RegulatorImportance[];
   # 相对常数基线的改进比例：1 - rmse / baselineRmse
   improvement: double;
   # 模型的种类：xgboost 或 constant
   kind: string;
   # 训练标签的取值范围上界（外推裁剪用）
   maxValue: double;
   # 训练标签的取值范围下界（外推裁剪用）
   minValue: double;
   # 被选中参与拟合的调控因子数量
   nRegulators: integer;
   # 训练集样本数
   nTrain: integer;
   # 验证集样本数（0 表示没有留出验证集）
   nValidate: integer;
   # 验证集上的决定系数 R²（没有验证集时退化为训练集 R²）
   r2: double;
   # 验证集上的均方根误差
   rmse: double;
   # 树的数量
   trees: integer;
}
