DiscoveryEdge

Description

一条被“残差二阶段”流程检验过的候选调控边。

Declare

            
# namespace SMRUCC.genomics.Analysis.GeneTree
export class DiscoveryEdge {
   # 靶基因模型残差与候选表达之间的 pearson 相关系数
   correlation: double;
   # 把这条边加入特征之后，验证集 R² 的增益
   deltaR2: double;
   isAccepted: boolean;
   # 加入这条边之后的验证集 R²
   r2After: double;
   # 加入这条边之前的验证集 R²
   r2Before: double;
   # 被拒绝的原因（接受时为空）
   reason: string;
   # 候选调控者
   regulator: string;
   # 该特征的偏依赖灵敏度
   sensitivity: double;
   # 该特征在最终模型之中被用作分裂特征的节点数量
   splits: integer;
   # 是否被接受为新调控边：accepted / rejected
   status: string;
   # 靶基因（被调控的一端）
   target: string;
}
