DeltaStarLevels

Description

For convenience, we describe levels of delta-differences for some reference examples (all values multiplied by 1000). (compatible rename of the legacy SimilarDiscriptions enum, follows the paper's delta* symbol)

# namespace SMRUCC.genomics.Analysis.SequenceTools.DNA_Comparative.DeltaSimilarity1998
export class DeltaStarLevels extends Enum {
   # (delta* <= 50; pervasively within species, human vs cow, Lactococcus lactis vs Streptococcus pyogenes).
   [@desc "delta* <= 50; pervasively within species, human vs cow, Lactococcus lactis vs Streptococcus pyogenes"]
   Close: DeltaStarLevels = 0;

   # (55 <= delta* <= 85; human vs chicken, Escherichia coli vs Haemophilus influenzae, Synechococcus vs Anabaena).
   [@desc "delta* = [55, 85]; human vs chicken, Escherichia coli vs Haemophilus influenzae, Synechococcus vs Anabaena"]
   ModeratelySimilar: DeltaStarLevels = 1;

   # (90 <= delta* <= 120; human vs sea urchin, M. genitalium vs M. pneumoniae).
   [@desc "delta* = [90, 120]; human vs sea urchin, M. genitalium vs M. pneumoniae"]
   WeaklySimilar: DeltaStarLevels = 2;

   # (125 <= delta* <= 145; human vs Sulfolobus, E. coli vs R. prowazekii, M. jannaschii vs M. thermoautotrophicum).
   [@desc "delta* = [125, 145]; human vs Sulfolobus, E. coli vs R. prowazekii, M. jannaschii vs M. thermoautotrophicum"]
   DistantlySimilar: DeltaStarLevels = 3;

   # (150 <= delta* <= 180; human vs Drosophila, E. coli vs Helicobacter pylori).
   [@desc "delta* = [150, 180]; human vs Drosophila, E. coli vs Helicobacter pylori"]
   Distant: DeltaStarLevels = 4;

   # (delta* >= 190; human vs E. coli, E. coli vs Sulfolobus, M. jannaschii vs Halobacterium).
   [@desc "delta* >= 190; human vs E. coli, E. coli vs Sulfolobus, M. jannaschii vs Halobacterium"]
   VeryDistant: DeltaStarLevels = 5;

}
