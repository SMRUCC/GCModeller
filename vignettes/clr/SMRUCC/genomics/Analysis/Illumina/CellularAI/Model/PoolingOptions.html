PoolingOptions {SMRUCC.genomics.Analysis.Illumina.CellularAI.Model} .NET clr documentation

PoolingOptions

Description

随机效应池化的参数：对应参考配置 model.pooling 段。

六个 *_row / *_col / *_mat 字符串是因子表达式：以 * 连接的因子名， 例如 "inv:mad:dl"（逆方差权重）、"sef*kd"（尺度误差 × 敲低系数）、 "auxiliary*sim*pair"（辅助行 × 相似度 × 配对降权）。 表达式的写法与相乘顺序都是模型定义的一部分（浮点乘法不可交换），因此这里保留为字符串， 由 FactorExpression 解析、由池化内核按序求值。

PoolingOptions.Views 决定参与池化的训练视图（按顺序），PoolingOptions.AuxiliaryWeight 是对 role = auxiliary 的整行加权倍数。

Declare

            
# namespace SMRUCC.genomics.Analysis.Illumina.CellularAI.Model
export class PoolingOptions {
   # 辅助行的整行加权倍数，默认 2。
   AuxiliaryWeight: double;
   # 池化后的修正方式（保留上游字段，默认 LZ）。
   Correction: string;
   # 缺失值填充方式（保留上游字段，默认 zf）。
   Fill: string;
   # 敲低系数的下限，默认 0（允许完全敲低）。
   KdFloor: double;
   # 敲低系数的参考值（归一化分母），默认 0.6。
   KdRef: double;
   # log2FC 的列（扰动）方向因子表达式，默认 sef*kd。
   L2Col: string;
   # log2FC 的整行因子表达式，默认 auxiliary*sim*pair。
   L2Mat: string;
   # log2FC 的行（基因）方向因子表达式，默认 inv:mad:dl。
   L2Row: string;
   # 噪声比因子的幂次，默认 1.0。
   Noise: double;
   # 是否使用单条证据流（保留上游字段，默认 True）。
   OneStream: boolean;
   # 配对/交替记录的降权倍数，默认 0.5。
   PairWeight: double;
   # 尺度误差因子的幂次，默认 1.0。
   Se: double;
   # 尺度误差比值的裁剪幅度，默认 4.0。
   SeClip: double;
   # 相似度权重的指数系数 β，默认 0.25。
   Sim: double;
   # 计算相似度时使用的高变基因数，默认 2000。
   SimHvg: integer;
   # 参与池化的训练视图，按顺序。
   Views: string;
   # z 通道的列方向因子表达式，默认 sef*kd。
   ZCol: string;
   # z 通道的整行因子表达式，默认 auxiliary*sim*pair。
   ZMat: string;
   # z 通道的行方向因子表达式，默认 nr（噪声比）。
   ZRow: string;
}
