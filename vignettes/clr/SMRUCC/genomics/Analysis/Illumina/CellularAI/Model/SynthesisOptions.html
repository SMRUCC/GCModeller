SynthesisOptions

Description

单细胞计数合成的参数：对应参考配置 model.synthesis 段。

SynthesisOptions.ZPlan 是计划基因的目标秩和 z 区间（按优先级从低到高线性插值）， SynthesisOptions.ZNudge 是非计划基因在微调强度 1 时的目标 z。

SynthesisOptions.Rmax / SynthesisOptions.NBis 控制 copula 相关系数的搜索范围与二分次数； SynthesisOptions.Ztol / SynthesisOptions.ZtolPlan 是"z 可以偏离目标多少"的容差（后者用于计划基因，更严）； SynthesisOptions.Ltol 是"平均每细胞 CPM 可以偏离放置目标多少"（对数尺度）。

SynthesisOptions.PlannedZ / SynthesisOptions.OvershootZ / SynthesisOptions.RepairR 决定"过冲修复"的触发条件与候选相关； SynthesisOptions.SolveNtop / SynthesisOptions.SolveTol / SynthesisOptions.SolveIters 控制平移二分； SynthesisOptions.ShiftBand 是平移的搜索区间；SynthesisOptions.SparsifyStep 是稀疏化扫描步长。

# namespace SMRUCC.genomics.Analysis.Illumina.CellularAI.Model
export class SynthesisOptions {
   # 平均 CPM 的对数容差，默认 0.01。
   Ltol: double;
   # 相关系数二分的次数，默认 5。
   NBis: integer;
   # 非表达基因突发散布的每块大小，默认 4。
   NeBurst: integer;
   # 过冲修复的触发幅度，默认 0.4。
   OvershootZ: double;
   # "计划基因"的 |z| 判定门槛，默认 5.0。
   PlannedZ: double;
   # 过冲修复的候选相关系数，默认 (0.25, 0.5)（另加其负值与 0、±rmax）。
   RepairR: double;
   # copula 相关系数的搜索上界，默认 0.99。
   Rmax: double;
   # 平移量的搜索区间，默认 [-0.6, 1.6]。
   ShiftBand: double;
   # 平移二分的最大迭代次数，默认 40。
   SolveIters: integer;
   # 质量守恒补足时使用的降序名次个数，默认 40。
   SolveNtop: integer;
   # 平移二分的 z 容差，默认 0.05。
   SolveTol: double;
   # 稀疏化扫描的步长，默认 0.025。
   SparsifyStep: double;
   # 非计划基因在满强度下的目标 |z|，默认 2.0。
   ZNudge: double;
   # 计划基因的目标 z 区间（低优先级 → 高优先级），默认 (5.5, 9.0)。
   ZPlan: double;
   # 非计划基因的 z 容差，默认 1.0。
   Ztol: double;
   # 计划基因的 z 容差，默认 0.2。
   ZtolPlan: double;
}
