SummaryRow

Description

一条训练摘要：某个细胞系、某个视图下的效应矩阵、概率矩阵与对照剖面。

这里刻意保留该行自己的基因轴与扰动轴（而不是提前对齐到任务轴）： 上游的 log2fc_stack 要在"训练基因的并集"上算相似度与方差， 提前裁剪到任务轴会改变高变基因的选择结果。

缺失用 NaN 表示（"未测"），与"测得 0"严格区分： 池化时未测项按"值为 0、权重照算"计入分母，而测得的 0 是真实观测。

# namespace SMRUCC.genomics.Analysis.Illumina.CellularAI.Summary
export class SummaryRow {
   # 细胞系（训练行）名称。
   CellLine: string;
   # 对照表达剖面。
   Control: ControlProfile;
   # log2FC 效应矩阵（该行自己的基因轴 × 扰动轴）。
   Effect: Double[][];
   # 效应矩阵的基因轴。
   EffectGenes: string;
   # 效应矩阵的扰动轴。
   EffectPerts: string;
   # 是否为辅助行。
   IsAuxiliary: boolean;
   # 行的唯一键 view/cell_line。
   Key: string;
   # 逐基因中心化表。
   Offset: OffsetTable;
   # 是否为配对组里的交替记录。
   PairAlternate: boolean;
   # 配对组名（同一实验的多个记录同名）；无配对时为 Nothing。
   PairGroup: string;
   # 双侧概率矩阵（与 SummaryRow.Effect 同形状）。
   Probability: Double[][];
   # 角色：primary 或 auxiliary。
   Role: string;
   # 可选的尺度矩阵（与 SummaryRow.Effect 同形状，用于 sef 因子）。
   Scale: Double[][];
   # 视图名称。
   View: string;
}
