# namespace SMRUCC.genomics.Analysis.Metagenome.MetaFunction.metaTraits.Traitar
export class TraitPrediction {
   # 预测的置信度，取值区间 [0,1]
   confidence: double;
   # 表型的数据类型（boolean / categorical / numeric (continuous)）
   data_type: string;
   # 主分类
   group_1: string;
   # 次分类
   group_2: string;
   # 预测结果：分类模型为预测的类别标签文本（如 true / rod）， 回归模型为预测数值的文本形式
   predict: string;
   # 决策函数的输出值（分类为边距，回归为回归值）
   score: double;
   # 该表型模型的状态：trained / skipped
   status: string;
   # 表型名称
   trait_name: string;
   # 计量单位
   unit: string;
   # 数值形式的结果：分类模型为类别的 factor 编码，回归模型为回归值
   value: double;
   # 每一个类别所获得的决策值（多分类投票）
   votes: double;
}
