ScreeningRecord

Description

一次虚拟扰动实验之中，某一个基因的响应记录

Declare

            
# namespace SMRUCC.genomics.Analysis.SGrid.Perturbation
export class ScreeningRecord {
   # 对照组在该基因之上的终末值
   ControlLevel: double;
   # 终末值的倍数变化（扰动组 / 对照组）
   FoldChange: double;
   # 观测的基因（响应者）
   Gene: string;
   # 该基因是否为本次扰动的直接目标
   IsTarget: boolean;
   # 整条轨迹之上相对对照组的最大绝对偏差
   MaxDeviation: double;
   # 扰动的类型描述
   Perturbation: string;
   # 扰动组在该基因之上的终末值
   PerturbedLevel: double;
   # 响应时间：轨迹第一次超出对照动态范围 5% 的时间点，负数表示没有任何响应
   ResponseTime: double;
   # 被扰动的目标基因
   Target: string;
}
