ExpressionTimeSeries {SMRUCC.genomics.Analysis.SGrid.GRN} .NET clr documentation

ExpressionTimeSeries

Description

The time series view of a gene expression matrix.(基因表达矩阵的时序视图)

本类型负责把 Matrix 之中的数据转换为可以被 S-system 逆问题所使用的形式： 行是基因、列是时间点，并且会预先计算出对数表达量矩阵以提升拟合时的求值性能。

Declare

            
# namespace SMRUCC.genomics.Analysis.SGrid.GRN
export class ExpressionTimeSeries {
   # 已经规整化之后的基因编号（可以直接被数学表达式引擎所解析）
   GeneNames: string;
   # 表达量的自然对数矩阵，第一维是基因、第二维是时间点
   LnX: Double[][];
   # 基因的数量
   nGenes: integer;
   # 建立该时序视图时所使用的归一化方法与伪计数
   Normalize: NormalizationMethods;
   # 时间点的数量
   nTimePoints: integer;
   # 每一个基因在预处理时被平移的常数 δ（幂律要求表达量为正数）： X = log2(max(raw,0) + pseudo) + δ。 把外部数据（例如 pseudobulk context）映射到模型空间时必须使用同一个 δ
   Offsets: double;
   # 原始的基因编号，用于结果标注
   OriginalNames: string;
   # 建立该时序视图时所使用的伪计数
   PseudoCount: double;
   # 时间轴的终点值
   TimeEnd: double;
   # 时间轴（每个采样点所对应的时间值）
   TimePoints: double;
   # 在预处理过程之中所产生的提示信息
   Warnings: iterates(String);
   # 表达量矩阵，第一维是基因、第二维是时间点
   X: Double[][];
}
