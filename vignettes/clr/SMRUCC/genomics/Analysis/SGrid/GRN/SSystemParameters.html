SSystemParameters {SMRUCC.genomics.Analysis.SGrid.GRN} .NET clr documentation

SSystemParameters

Description

The fitted parameter set of the S-system model. (拟合得到的S-system参数集合：alpha / beta / G / H)

其中 SSystemParameters.G(i)(j) 为合成项之中基因 j 对基因 i 的调控弹性 g_ij， SSystemParameters.H(i)(j) 为降解项之中的 h_ij；净调控弹性取 g_ij - h_ij。

Declare

            
# namespace SMRUCC.genomics.Analysis.SGrid.GRN
export class SSystemParameters {
   # 每一个基因的合成速率常数 alpha
   alpha: double;
   # 每一个基因的降解速率常数 beta
   beta: double;
   # 每一个基因的候选调控因子（矩阵行号）；为空表示该基因使用了全部基因作为候选
   Candidates: Int32[][];
   # 每一个基因的参数求解是否收敛
   Converged: boolean;
   # 合成项的动力学阶矩阵，第一维是靶基因、第二维是调控因子
   G: Double[][];
   # 规整化之后的基因编号
   GeneNames: string;
   # 降解项的动力学阶矩阵，第一维是靶基因、第二维是调控因子
   H: Double[][];
   # 全部基因的平均拟合均方根误差
   MeanRMSE: double;
   # 基因的数量
   nGenes: integer;
   # 原始的基因编号
   OriginalNames: string;
   # 每一个基因的斜率拟合均方根误差
   SlopeRMSE: double;
   # 每一个基因的斜率拟合残差平方和
   SlopeSSE: double;
   # 每一个基因的观测斜率的总平方和（用于计算拟合的 R²）
   SlopeSST: double;
   # 是否执行过第二阶段的轨迹精修
   TrajectoryRefined: boolean;
}
