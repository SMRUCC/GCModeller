MultiOmicsGsvaResult {SMRUCC.genomics.Analysis.HTS.GSVA} .NET clr documentation

MultiOmicsGsvaResult

Description

多组学 GSVA 的完整结果。

三个标准输出对齐 method.md 的建议： (1) 样本 × 通路评分矩阵 MultiOmicsGsvaResult.Scores； (2) 通路 × 组学贡献分解 MultiOmicsGsvaResult.Contributions / MultiOmicsGsvaResult.ContributionShare； (3) 通路显著性 p 值表 MultiOmicsGsvaResult.PValues / MultiOmicsGsvaResult.QValues。

Declare

            
# namespace SMRUCC.genomics.Analysis.HTS.GSVA
export class MultiOmicsGsvaResult {
   # 跨组学符号一致度（通路 × 样本）
   Consistency: Matrix;
   # 逐通路的组学贡献 c_k = w_k · ES_k
   Contributions: list;
   # 逐通路的组学贡献占比
   ContributionShare: list;
   # 逐通路的组学成员数 |S_k|
   Coverage: list;
   # 每个样本上主导的组学层
   DominantOmics: list;
   # 总耗时（毫秒）
   ElapsedMilliseconds: integer;
   # 逐组学分项 ES_k（omics → 通路 × 样本）
   LayerScores: list;
   # 参与计算的组学层
   OmicsNames: string;
   # 本次运行使用的配置
   Options: MultiOmicsGsvaOptions;
   # 通路顺序
   Pathways: string;
   # 拼接矩阵消融对照（未计算时为 Nothing）
   Pooled: PooledBaselineResult;
   # 逐通路 p 值（未启用置换时为 Nothing）
   PValues: list;
   # 逐通路 BH 校正后的 q 值
   QValues: list;
   # 加权平均分数 ES_raw（未施加一致性奖励）
   RawScores: Matrix;
   # 样本轴
   SampleID: string;
   # 融合后的通路活性矩阵 ES_multi（通路 × 样本）
   Scores: Matrix;
   # 置换检验的完整结果（未启用时为 Nothing）
   Significance: PermutationResult;
   # 逐通路被跳过的组学层
   SkippedLayers: list;
   # 逐通路的组学权重
   Weights: list;
}
