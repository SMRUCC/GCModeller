# namespace SMRUCC.genomics.Analysis.HTS.GSVA
export class MultiOmicsGsvaOptions {
   # 是否叠加「组学特征数平衡项」1/log2(2+|S_k|)，用于抑制成员多的组学主导融合结果。
   BalanceFeatureSize: boolean;
   # 是否计算「拼接矩阵直接做单组学 GSVA」的消融对照
   ComputePooled: boolean;
   # 跨组学一致性奖励项的超参 λ（0 表示关闭，等价于纯粹的加权平均基线）。
   ConsistencyLambda: double;
   # 组学加权融合策略。
   FusionStrategy: FusionMethods;
   # 核密度估计类型：KCDFs.Gaussian（连续数据，默认）/ KCDFs.Poisson（计数数据）/ KCDFs.none（经验 CDF）。
   Kcdf: Nullable`1;
   # 未显式指定通路子集时，参与置换检验的最大通路数
   MaxPermutationPathways: integer;
   # 基因集最大尺寸（成员数，映射到特征之后）
   MaxSize: integer;
   # 基因集最小尺寸（成员数，映射到特征之后）
   MinSize: integer;
   # 置换检验视图中保留的特征数上限（0 表示不裁剪）。 裁剪时优先保留目标通路的成员，再用固定种子补充背景特征。
   PermutationGeneLimit: integer;
   # 置换检验使用的打分方法，默认 ssGSEA（基于秩，避免重复做核密度估计）。
   PermutationMethod: Methods;
   # 参与置换检验的通路子集；为空时取融合结果中的前 MultiOmicsGsvaOptions.MaxPermutationPathways 条通路。
   PermutationPathways: string;
   # 置换检验次数（0 表示跳过显著性评估）。
   Permutations: integer;
   # 组学内部的富集打分方法，推荐 Methods.gsva（默认）。
   ScoreMethod: Methods;
   # 随机种子（派生数据、置换打乱均使用它保证可复现）
   Seed: integer;
   # ssGSEA 打分路径的 alpha 参数
   SsGseaAlpha: double;
   # ssGSEA 打分路径是否做分数归一化
   SsGseaNormalize: boolean;
   # 是否输出过程日志
   Verbose: boolean;
}
