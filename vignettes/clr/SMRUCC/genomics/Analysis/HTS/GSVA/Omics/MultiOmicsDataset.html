MultiOmicsDataset

Description

K 个组学层的容器：统一样本轴 + 特征三元索引 + 拼接视图。

对应 method.md 步骤1：输入为拼接矩阵 X ∈ R^{(N1+N2+...+NK)×M}， 特征标签表保留 (feature_id, omics_source, gene)； 但主流程不会真的把矩阵拼接后做一次 GSVA（见 PooledBaseline 的消融对照）， 拼接视图只用于对外导出特征表与消融实验。

# namespace SMRUCC.genomics.Analysis.HTS.GSVA.Omics
export class MultiOmicsDataset {
   # 因为并非所有层都含有而被丢弃的样本
   DroppedSamples: string;
   # 所有层特征数之和（即拼接矩阵的行数）
   FeatureCount: integer;
   # 拼接矩阵的特征表
   FeatureTable: OmicsFeature[];
   LayerCount: integer;
   # 已按统一样本轴对齐的组学层
   Layers: OmicsLayer[];
   OmicsNames: string;
   # 样本数量
   SampleCount: integer;
   # 跨组学共享的样本轴（按第一个层的原始顺序）
   SampleID: string;
}
