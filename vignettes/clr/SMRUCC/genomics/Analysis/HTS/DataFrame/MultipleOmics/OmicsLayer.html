OmicsLayer

Description

单个组学层的「特征 × 样本」矩阵模型。

每个组学层独立持有自己的特征集合与样本向量，组学之间只在样本轴上对齐， 特征轴保持独立（这正是"保留组学来源"的实现基础）。

Declare

            
# namespace SMRUCC.genomics.Analysis.HTS.DataFrame.MultipleOmics
export class OmicsLayer {
   # 方向约定：+1 表示数值升高代表通路活性升高（mRNA / protein）， -1 表示数值升高代表通路活性下降（例如甲基化 beta 值）， 打分完成后会对该层的 ES 取负以统一到"活性方向"。
   Direction: integer;
   # 原始（或经过层内标准化之后的）表达/丰度矩阵，行是特征、列是样本。
   Expr: Matrix;
   # 特征（行）数量。
   FeatureCount: integer;
   # 与 OmicsLayer.Expr 行一一对应的特征 id 向量（基因符号）。
   Genes: string;
   # 组学层名称，例如 mRNA / methylation / protein。
   Name: string;
   # 当前矩阵已经施加过的层内标准化方法名，仅用于溯源与日志。
   Normalization: string;
   # 样本（列）数量。
   SampleCount: integer;
   # 与 OmicsLayer.Expr 列一一对应的样本 id 向量。
   SampleID: string;
}
