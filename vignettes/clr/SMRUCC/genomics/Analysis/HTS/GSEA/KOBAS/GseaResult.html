GseaResult

Description

一个基因集的 GSEA 分析结果。

Declare

            
# namespace SMRUCC.genomics.Analysis.HTS.GSEA.KOBAS
export class GseaResult {
   # 核心富集基因（峰前或峰后的命中基因）。
   CoreGenes: string;
   # 基因集描述。
   Description: string;
   # 观测到的富集分数。
   ES: double;
   # 富集分数在 running sum 序列之中的位置。
   ESIndex: integer;
   # 错误发现率；跳过置换检验时为 Double。
   FDR: double;
   # 该基因集在排序列表之中的全部命中基因。
   HitGenes: string;
   # 基因集编号（GMT 第一列）。
   ID: string;
   # 基因集名称（GMT 第二列）。
   Names: string;
   # 归一化富集分数；跳过置换检验时等于 GseaResult.ES。
   NES: double;
   # 名义 p 值；跳过置换检验时为 Double。
   PValue: double;
   # running sum 序列；当关闭 GseaOptions.KeepRunningSum 时为 Nothing。
   RES: double;
   # 该基因集在排序列表之中的命中数量。
   Size: integer;
}
