# namespace SMRUCC.genomics.Analysis.FuzzyGene.Fitting
export class FuzzyGeneTrainingResult {
   # 拟合阶段的总分配字节数（MB）；用于衡量缓冲复用的效果。
   AllocatedMB: double;
   # 导出的带符号调控边数量。
   EdgeCount: integer;
   # 耗时。
   Elapsed: TimeSpan;
   # 拟合阶段的 Gen2 GC 次数；大对象分配是否被消除，看这个最直接。
   Gen2Collections: integer;
   # 逐基因诊断记录。
   GeneMetrics: GeneFitMetric[];
   # 成功基因的平均决定系数。
   MeanR2: double;
   # 成功基因的平均训练 RMSE。
   MeanRmse: double;
   # 实际采用的拟合路线。
   Mode: FuzzyTargetMode;
   # 相关性筛选实际使用的候选基因数量。
   NCandidatePool: integer;
   # 训练失败的基因数。
   NFailed: integer;
   # 参与建模的基因总数。
   NGene: integer;
   # 训练样本（状态行）数量（行采样之后）。
   NSample: integer;
   # 原始样本数量（行采样之前）。
   NSampleRaw: integer;
   # 被跳过的基因数（常表达、缺少候选因子等）。
   NSkipped: integer;
   # 成功训练出模型的基因数。
   NTrained: integer;
   # 实际使用的并行 worker 线程数（诊断 128 核机器是否真的跑满）。
   ParallelWorkers: integer;
   # 预处理（统计量 + 列构建 + 列统计）耗时。
   PreprocessElapsed: TimeSpan;
   # 候选筛选累计耗时（CPU 时间，跨所有并行线程求和）。
   SelectionElapsed: TimeSpan;
   # 单基因训练累计耗时（CPU 时间，跨所有并行线程求和）。
   TrainingElapsed: TimeSpan;
   # 降级、异常与数据质量告警。
   Warnings: string;
}
