虚拟扰动的效应传播方式。

# namespace SMRUCC.genomics.Analysis.CellPhenotype
export class PropagationMode extends Enum {
   # 沿调控边级联迭代直到稳态(默认)。
   # 
   # 被扰动的基因在每一轮迭代之中都会被强制回设到干预值(do 算子语义)， 因此它不会被自己的上游调控因子"修复"回去；其余基因按照 x_t = c_t + sum(beta * x_r) 的 Jacobi 迭代更新。
   # 
   # 这是贝叶斯网络引擎"按拓扑序一次性联合采样"在有环线性图上面的等价物。
   Cascade: PropagationMode = 0;

   # 只计算一阶(直接)效应，不进行级联迭代。
   # 
   # 只有被扰动基因的直接靶基因会产生变化，等价于 delta_target = beta * delta_regulator。结果保守且绝对不会发散。
   OneHop: PropagationMode = 1;

}
