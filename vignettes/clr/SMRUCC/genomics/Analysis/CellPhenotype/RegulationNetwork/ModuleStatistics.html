单个 WGCNA 模块在构建过程中的统计信息

Declare

            
# namespace SMRUCC.genomics.Analysis.CellPhenotype.RegulationNetwork
export class ModuleStatistics {
   # 通过 |cor| 阈值的候选边数
   candidateEdges: integer;
   # 由 TF 注释成功定向的边数
   directedEdges: integer;
   # 被 DPI 判定为间接而去除的边数
   dpiRemovedEdges: integer;
   # 该模块的耗时（毫秒）
   elapsedMilliseconds: double;
   # 通过 FDR 阈值的边数
   fdrPassedEdges: integer;
   # 模块基因数（规模控制截断前的原始基因数）
   geneCount: integer;
   # WGCNA 模块名称（通常为模块颜色）
   moduleName: string;
   # 被偏相关判定为间接而去除的边数
   partialRemovedEdges: integer;
   # 仅由 STRING 新增的边数
   stringOnlyEdges: integer;
   # 得到 STRING 证据支持的边数
   stringSupportedEdges: integer;
   # 无法定向、作为候选保留的边数（TF-TF / 非 TF-非 TF）
   undirectedEdges: integer;
   # 实际参与计算的基因数（规模控制截断后）
   usedGenes: integer;
}
