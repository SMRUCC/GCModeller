Definition {SMRUCC.genomics.GCModeller.ModellingEngine.Dynamics.Definitions} .NET clr documentation

Definition

Description

The definition for the mass environment for run the simulation, use for context mapping across database

因为物质编号可能会来自于不同的数据库，所以会需要使用这个对象将一些关键的物质映射为计算引擎所能够被识别的对象

Declare

            
# namespace SMRUCC.genomics.GCModeller.ModellingEngine.Dynamics.Definitions
export class Definition {
   ADP: string;
   # amino acid metabolite id mapping for construct of the polypeptide in translation process
   AminoAcid: AminoAcid;
   Ammonia: string;
   ATP: string;
   # 氧化还原载体与酶辅因子映射：[辅因子名称 => 化合物编号]
   # 
   # 常用键：NAD、NADH、NADP、NADPH、FAD、FADH2、CO-A、ACETYL-COA、 THIAMINE-PP、PYRIDOXAL-P、COBALAMIN
   Cofactors: list;
   # define of the cell initial status.
   context: CellContext[];
   # the compartment id of the Culture medium
   CultureMedium: string;
   # 默认培养基配方。
   # 
   # 一个组分由「角色(role) + 数据库化合物编号(id) + 默认浓度(mM)」构成， 因此同一个配方可以在 MetaCyc / BioCyc / KEGG 之间切换而无需改动调用代码。 模拟程序在未指定培养基文件的时候，会按本表初始化胞外环境，并且只保留 模型之中真实存在的化合物编号。
   CultureMediumComponents: MediumComponent[];
   GDP: string;
   # define the id mapping to some general compounds
   GenericCompounds: list;
   GTP: string;
   # 矿物离子映射：[离子符号 => 化合物编号]
   # 
   # 常用键：K、Na、Mg、Ca、Cl、Fe2、Fe3、Zn、Mn、Cu、Co、Ni、Mo、H、OH
   Ions: list;
   # nucleotide metabolite id mapping for construct of the RNA molecule in the transcription process.
   NucleicAcid: NucleicAcid;
   Oxygen: string;
   PI: string;
   PPI: string;
   transmembrane: Transmembrane;
   # 分泌产物 / 废物映射：[产物名称 => 化合物编号]
   # 
   # 常用键：CO2、HCO3、LACTATE、ETHANOL、FORMATE、ACETATE、H2、H2S、NH3
   WasteProducts: list;
   Water: string;
}
