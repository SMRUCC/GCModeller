# namespace SMRUCC.genomics.Model.Network.Regulons.LinearRegression
export class ElasticNetMode extends Enum {
   # 固定 l1_ratio，求解整条正则化路径之后，在「非零系数数量不超过每基因上限」 的步骤之中挑选 R^2 最高的那一步。
   # 
   # 这是最快的模式，单个基因的成本与 LASSO 的正则化路径求解相当。
   RegularizationPath: ElasticNetMode = 0;

   # 每一个目标基因都单独做一次 K 折交叉验证，自动选择 (alpha, l1_ratio)。
   # 
   # 这是最准确的模式，但是单个基因的成本大约是路径模式的 l1_ratio 网格数量 × alpha 步数 × 折数 倍，只适合基因数量很少的数据。
   PerGeneCrossValidation: ElasticNetMode = 1;

   # 先挑选若干代表性基因(表达方差最大的那些)做一次交叉验证， 取超参数的中位数供全部基因复用，然后逐基因做单点拟合。
   # 
   # 精度与成本之间的折中方案，推荐在基因数量很多的数据上面使用。
   GlobalCrossValidation: ElasticNetMode = 2;

}
