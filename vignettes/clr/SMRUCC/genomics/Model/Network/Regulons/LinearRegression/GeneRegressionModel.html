GeneRegressionModel {SMRUCC.genomics.Model.Network.Regulons.LinearRegression} .NET clr documentation

GeneRegressionModel

Description

单个目标基因的线性拟合模型

 target = intercept + beta_1 * regulator_1 + beta_2 * regulator_2 + ...

其中 GeneRegressionModel.regulators 与 GeneRegressionModel.weights 是一一对应的， 第 i 个回归系数描述第 i 个调控因子对当前目标基因的表达贡献。

Declare

            
# namespace SMRUCC.genomics.Model.Network.Regulons.LinearRegression
export class GeneRegressionModel {
   # 弹性网络所选用的总正则化强度，其它求解方法之下这个数值为零。
   alpha: double;
   # 残差的自由度，即 样本数量 - 自变量个数
   df: integer;
   # 拟合失败的时候的异常信息，求解成功的时候为空字符串
   errorMessage: string;
   # 模型的常数项(截距)，即用户在公式之中写出来的 const。
   # 
   # LASSO 求解器自身给出了标准化还原之后的截距，因此两种方法之下都有常数项。
   intercept: double;
   # 截距项的显著性 p 值
   interceptPvalue: double;
   # 截距项的 t 值
   interceptT: double;
   # 当前模型是否求解成功
   IsValid: boolean;
   # 弹性网络所选用的 L1 混合比例，其它求解方法之下这个数值为零。
   l1Ratio: double;
   # LASSO 模型在正则化路径上面所处的位置对应的 lambda 值， 其它求解方法之下这个数值为零。
   lambda: double;
   # 当前模型所使用的求解方法
   method: FittingMethod;
   # 参与拟合的样本数量
   N: integer;
   # 模型之中非零或者显著的调控因子的数量
   nonZeroWeights: integer;
   # 每一个回归系数对应的显著性 p 值，无法计算的时候为空
   pvalues: double;
   # 拟合优度 R^2
   r2: double;
   # 参与拟合的调控因子基因编号，顺序与 GeneRegressionModel.weights 一致。
   regulators: string;
   # 拟合的均方根误差
   rmse: double;
   # 残差平方和
   sse: double;
   # 总平方和
   sst: double;
   # 目标基因的编号
   target: string;
   # 每一个回归系数对应的 t 值，无法计算的时候为空
   tvalues: double;
   # 每一个调控因子的回归系数
   weights: double;
}
