# namespace SMRUCC.genomics.Model.Network.Regulons.ARACNE
export class MiThreshold {
   # 零分布所使用的互信息估计器
   estimator: MiEstimator;
   # 阈值的确定方式
   mode: MiThresholdMode;
   # 置换零分布的最大互信息
   nullMax: double;
   # 置换零分布的平均互信息(也就是“没有任何调控关系”时的估计偏置)
   nullMean: double;
   # 置换零分布的标准差
   nullSD: double;
   # 零分布所取的分位数(MiThresholdMode.Fixed 模式之下没有意义)
   quantile: double;
   # 每一个基因对被洗牌的次数
   replicates: integer;
   # 参与零分布估计的基因对数量
   samples: integer;
   # 显著性倍率：真实边所要求的互信息至少是零分布平均值的多少倍。
   # 
   # 这个数值用于判断阈值是否“过于宽松”(接近零分布的噪声水平)， 小于 1 说明当前的阈值完全落在噪声区间之内。
   SignalRatio: double;
   # 最终使用的互信息阈值：互信息低于这个数值的边会被过滤掉
   value: double;
}
