AracneStatistics

Description

一次 ARACNE 推断过程的统计信息

Declare

            
# namespace SMRUCC.genomics.Model.Network.Regulons.ARACNE
export class AracneStatistics {
   # 计算了互信息的基因对数量
   candidatePairs: integer;
   # DPI 的详细统计
   dpi: DpiResult;
   # 赋予方向之后的调控边数量
   edges: integer;
   # DPI 过滤之后的边数量(无向)
   edgesAfterDpi: integer;
   # 阈值过滤以及 top-K 剪枝之后的边数量
   edgesAfterThreshold: integer;
   # 阈值、DPI 与定向阶段的耗时(秒)
   filterSeconds: double;
   # 预处理之后实际参与计算的基因数量
   genes: integer;
   # 表达矩阵之中被扫描的基因数量
   genesScanned: integer;
   # 平均每个基因对的互信息计算耗时(微秒)
   microSecondsPerPair: double;
   # 互信息计算阶段的耗时(秒)
   miSeconds: double;
   # 预处理阶段的耗时(秒)
   preprocessSeconds: double;
   # curated 先验网络之中落在本次计算空间(基因集合)之内的调控关系数量
   priorEdges: integer;
   # 落在本次计算空间之内的先验边，按照置信度分级的数量。
   # 
   # 与 AracneStatistics.priorRescuedByConfidence 相除即可得到每一档置信度 对应的“保护成功率”，也就是分级保护是否真的按置信度倾斜的量化证据。
   priorPairsByConfidence: list;
   # 被先验保护的边数量：这些边满足 DPI 的剔除条件， 但是因为 curated 先验支持而保留了下来
   priorProtectedEdges: integer;
   # 本次推断所实际使用的先验保护方式
   priorProtection: PriorProtectionMode;
   # 被保护(从 DPI 之中救回来)的边按照先验置信度分级的统计： 键名为 high / medium / low。
   # 
   # 这是验证“分级保护是否生效”的直接证据：Graded 模式下 high 组应当明显多于 low 组。
   priorRescuedByConfidence: list;
   # 最终输出之中受先验支持的调控边数量(有向边的口径，TF-TF 的 mutual 边会被计算两次)
   priorSupportedEdges: integer;
   # 最终输出之中受先验支持的无向基因对数量。
   # 
   # 与 AracneStatistics.priorEdges 相除就得到先验网络在本次推断结果之中的召回率。
   priorSupportedPairs: integer;
   # 样本(细胞)数量
   samples: integer;
   # 互信息的显著性阈值
   threshold: MiThreshold;
   # 整个推断过程的耗时(秒)
   totalSeconds: double;
   # 匹配到的转录因子数量
   transcriptionFactors: integer;
}
