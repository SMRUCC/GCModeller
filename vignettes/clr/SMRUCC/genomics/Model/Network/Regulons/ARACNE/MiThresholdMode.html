互信息显著性阈值 tau 的确定方式。

Declare

            
# namespace SMRUCC.genomics.Model.Network.Regulons.ARACNE
export class MiThresholdMode extends Enum {
   # 置换零分布(默认)：把每一个基因的表达向量沿着样本维度独立地随机打乱， 破坏基因之间的依赖关系但是保留各自的边缘分布，然后重新计算互信息， 得到“没有任何调控关系”的零分布；取其分位数(默认 95%)作为 tau。
   # 
   # 这样就可以避免硬编码阈值：不同数据集的样本数量、离散化粒度、 估计器偏置都会被零分布同样地“抵消”掉一部分。
   Permutation: MiThresholdMode = 0;

   # 使用 AracneOptions.miThreshold 之中给定的固定阈值。
   Fixed: MiThresholdMode = 1;

}
