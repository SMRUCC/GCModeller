# namespace SMRUCC.genomics.Analysis.PanGenome.ReportJSON
export class SVDomainEntropyPoint {
   # KMeans 聚类结果（按照簇中心确定性重排之后的簇编号）
   cluster: integer;
   # 该家族的 SV CopyNumber 矩阵行（跨基因组的拷贝数分布）的香农信息熵
   hCopyNumber: double;
   # 该家族的 SV Median 矩阵行的香农信息熵（按Median的精确取值统计频率）
   hMedian: double;
   # 基因家族ID
   name: string;
   # 该家族存在SV事件的基因组数量
   presentGenomes: integer;
   # Z-score 标准化之后的 CopyNumber 熵（散点图的X轴）
   zCopyNumber: double;
   # Z-score 标准化之后的 Median 熵（散点图的Y轴）
   zMedian: double;
}
