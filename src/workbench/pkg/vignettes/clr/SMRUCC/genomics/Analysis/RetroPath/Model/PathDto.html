PathDto

Description

一条候选通路的 JSON DTO：评分明细 + 正向生物合成步骤。

Declare

            
# namespace SMRUCC.genomics.Analysis.RetroPath.Model
export class PathDto {
   # 整条通路的 ΔG 合计（kJ/mol）。
   DeltaGTotal: double;
   # 酶可得性得分 (0,1]。
   EnzymeScore: double;
   # 加权全局得分，用于排序。
   GlobalScore: double;
   # 通路编号，形如 "path_1"（按全局分降序编号）。
   Id: string;
   # 长度得分 = 1 / 步数。
   LengthScore: double;
   # 反应步数。
   NumSteps: integer;
   # 正向生物合成顺序的步骤列表（汇前体 → 目标）。
   Steps: ForwardStepDto[];
   # 热力学可行性得分 (0,1)。
   ThermoScore: double;
}
