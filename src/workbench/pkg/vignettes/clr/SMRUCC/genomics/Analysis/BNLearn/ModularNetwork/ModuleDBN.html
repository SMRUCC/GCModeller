单个 WGCNA 模块的 DBN 子网络训练结果容器（模块内部使用，不污染公共 API）。

# namespace SMRUCC.genomics.Analysis.BNLearn.ModularNetwork
export class ModuleDBN {
   # 模块 eigengene 轨迹：各时间点的模块基因均值向量（用于模块间相关度）
   Eigengene: double;
   # 模块内基因名 → 在 Genes 数组中的索引
   GeneIndex: list;
   # 该模块参与训练的基因名（与 timeSeries 子矩阵对齐）
   Genes: string;
   # WGCNA 模块颜色标签
   ModuleColor: string;
   # 已构建拓扑并完成参数学习的动态贝叶斯网络子网络
   Net: DynamicBayesianNetwork;
   # 该模块各基因的野生型（未受扰动）表达丰度：基因 ID → 丰度值。
   # 
   # 由训练流程自动计算（各基因在时间序列上的中位数，对 dropout 零值与 异常值稳健），作为虚拟扰动推演的初始状态与响应参照基准。早期实现把初始 状态硬编码为"全部 Medium"，导致未受扰动影响的基因一律输出 Medium。 可由 BlockBayesianNetwork.SetWildtypeBaseline 覆盖。
   WildtypeAbundance: list;
}
